PEP 8 Compliant Version

The final version adds a docstring and descriptive variable names, resolving every bug the human and AI reviews found.

def convert_temp(celsius):
    """Print a Celsius temperature converted to Fahrenheit and Kelvin."""
    fahrenheit = celsius * 9 / 5 + 32
    kelvin = celsius + 273.15
    print('Celsius:', celsius)
    print('Fahrenheit:', fahrenheit)
    print('Kelvin:', kelvin)


temp = float(input('Enter temp in Celsius: '))
convert_temp(temp)
